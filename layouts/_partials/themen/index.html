{{- /* Index der Themen (content/themen/{slug}/_index.md), einmal pro Build (partialCached). "sachgebiete/katalog.html" . -}} {{- $themen := slice -}} {{- $nachNorm := dict -}} {{- $nachSachgebiet := dict -}} {{- with site.GetPage "/themen" -}} {{- range sort .Sections "Title" -}} {{- $p := . -}} {{- $slug := path.Base .Path -}} {{- $hgs := slice -}} {{- range $tag := .Params.sachgebiete -}} {{- with index $katalog.eintraege $tag -}} {{- $hgs = $hgs | append .hauptgebiet -}} {{- $liste := (index $nachSachgebiet .sachgebiet) | default slice -}} {{- $nachSachgebiet = merge $nachSachgebiet (dict .sachgebiet ($liste | append (dict "page" $p "key" $tag))) -}} {{- end -}} {{- end -}} {{- $normen := slice -}} {{- range .Params.normen -}} {{- $key := printf "%s/%s" .gesetz .art -}} {{- $g := site.GetPage (printf "/kommentar/%s" .gesetz) -}} {{- $rest := strings.TrimPrefix "art-" .art $nachNorm (dict $key ($liste | append $p)) -}} {{- end -}} {{- $verifiziert := false -}} {{- with .Params.revisions -}} {{- $verifiziert = and ($p.Params.agent_verified | default false) ((index . "querschnitt" (gt (len (uniq $hgs)) 1) "normen" $normen "verifiziert" $verifiziert ) -}} {{- end -}} {{- end -}} {{- return dict "themen" $themen "nachNorm" $nachNorm "nachSachgebiet" $nachSachgebiet
